/*
Given an array of intervals where intervals[i] = [starti, endi], merge all overlapping intervals, and return an array of the non-overlapping intervals that cover all the intervals in the input.

 
Example 1:
Input: intervals = [[1,3],[2,6],[8,10],[15,18]]
Output: [[1,6],[8,10],[15,18]]
Explanation: Since intervals [1,3] and [2,6] overlap, merge them into [1,6].

Example 2:

Input: intervals = [[1,4],[4,5]]
Output: [[1,5]]
Explanation: Intervals [1,4] and [4,5] are considered overlapping.


Example 3:

Input: intervals = [[1,2],[1,6],[3,5],[8,10]]
Output: [[1,5]]
Explanation: Intervals [1,4] and [4,5] are considered overlapping.


1 - iterarte an array from end
2 - put the PQ ()

*/
public class Solution {
    
    
    public static List<Integer[]> mergeIntervals(List<Integer[]> intervals) {
        PriorityQueue<Integer[]> pq = new PriorityQueue<>();
        //reverse iteration
        for(int i = intervals.size() - 1; i >= 0; i--) {
            if(pq.isEmpty()) {
                pq.add(intervals.get(i));
            }else {
                Integer[] curr = intervals.get(i); //[3,5]
                Integer[] last = pq.peek(); //[8,10]
                // not merable
                if(curr[1] < last[0]) {
                    pq.add(curr);
                } else {
                    pq.remove(last); //[3,5]
                    Integer[] merge = new Integer[] {Math.min(curr[0], last[0]), Math.max(curr[1], last[1])};
                    pq.add(merge);
                }
            }
        }
        
        List<Integer[]> ans = new ArrayList<>();
        
        while(!pq.isEmpty()) {
            ans.add(pq.poll());
        }
        
        return ans;
    }

    public static void main(String[] args) {
        // [[1,3],[2,6],[8,10],[15,18]]
        List<Integer[]> intervals = Arrays.asList(new Integer[]{1,3}, new Integer[]{2,6},new Integer[]{8,10}, new Integer[]{15,18});
        System.out.println(mergeIntervals(intervals));
    }
}

  public static List<Integer> validatePassword(String password) {
    List<Integer> result = new ArrayList<>();
    // 1. It has to be at least 16 characters long.
    if(password.length() < 16) {
      result.add(1);
    }
    
    // 2. The password cannot contain the word "password". This rule is not case-sensitive.
    if (password.toLowerCase().contains("password")) {
      result.add(2);
    }
    
    // 3. The same character cannot be used more than 4 times. This rule is case-sensitive, "a" and "A" are different characters.
    boolean hasLowerCase = false;
    boolean hasUpperCase = false;
    boolean hasSpecialChar = false;
    Map<Character, Integer> map = new HashMap<>();
    for(int i = 0; i < password.length(); i++) {
      char cChar = password.charAt(i);
      map.put(cChar, map.getOrDefault(cChar, 0) + 1);
            
      if(!hasLowerCase && Character.isLowerCase(cChar)) {
        hasLowerCase = true;
      }
      
      if(!hasUpperCase && Character.isUpperCase(cChar)) {
        hasUpperCase = true;
      }
      
      if(!hasSpecialChar && (cChar == '*' || cChar == '#' || cChar == '@')) {
        hasSpecialChar = true;
      }
    }
    
    for(char key : map.keySet()) {
      if(map.get(key) > 4) {
        result.add(3);
        break;
      }
    }
    
    // 4. The password has to contain at least one uppercase and one lowercase letter.
    if(!hasLowerCase || !hasUpperCase) {
      result.add(4);
    }
    
    // 5. The password has to contain at least one of the following special characters "*","#","@".
    if(!hasSpecialChar) {
      result.add(5);
    }
    
    return result;
  }
